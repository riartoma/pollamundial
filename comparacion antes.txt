    <?php
    function calcularPuntos($pred_local, $pred_visit, $real_local, $real_visit) {
        if ($pred_local == $real_local && $pred_visit == $real_visit) return 5;  // Exacto
        
        $pred_ganador = $pred_local > $pred_visit ? 'local' : ($pred_local < $pred_visit ? 'visit' : 'empate');
        $real_ganador = $real_local > $real_visit ? 'local' : ($real_local < $real_visit ? 'visit' : 'empate');
        
        if ($pred_ganador != $real_ganador) return 0;
        
        if ($pred_ganador == 'empate') return 3;
        
        $pred_diff = abs($pred_local - $pred_visit);
        $real_diff = abs($real_local - $real_visit);
        $off = abs($pred_diff - $real_diff);
        
        if ($off == 0) return 4;
        if ($off == 1) return 3;
        if ($off == 2) return 2;
        return 1;  // Ganador correcto pero off >2
    }
    
    $pdo = conectarDB();
    
    if (isset($_POST['calcular'])) {
        // Limpiar puntuaciones viejas
        $pdo->query("DELETE FROM puntuaciones");
        
        // Obtener todos los resultados reales
        $stmt_real = $pdo->query("SELECT partido_id, goles_local, goles_visitante FROM resultados_reales");
        while ($real = $stmt_real->fetch()) {
            $partido_id = $real['partido_id'];
            
            // Predicciones por partido
            $stmt_pred = $pdo->prepare("SELECT usuario_id, goles_local, goles_visitante FROM predicciones WHERE partido_id = ?");
            $stmt_pred->execute([$partido_id]);
            while ($pred = $stmt_pred->fetch()) {
                $puntos = calcularPuntos($pred['goles_local'], $pred['goles_visitante'], $real['goles_local'], $real['goles_visitante']);
                
                // Guardar puntuación
                $stmt_punt = $pdo->prepare("INSERT INTO puntuaciones (usuario_id, partido_id, puntos) VALUES (?, ?, ?)");
                $stmt_punt->execute([$pred['usuario_id'], $partido_id, $puntos]);
                
                // Actualizar total usuario
                $pdo->prepare("UPDATE usuarios SET puntos_total = puntos_total + ? WHERE id = ?")->execute([$puntos, $pred['usuario_id']]);
            }
        }
        echo "<p>Cálculo completado!</p>";
    }
